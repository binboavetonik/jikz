% Scopes, loops, calc, and the environment.
\begin{tikzpicture}[scale=1, >=stealth, node distance=1.5cm, every node/.style={draw}]
  \begin{scope}[shift={(1,1)}]
    \draw (0,0) -- (1,0);
  \end{scope}
  \begin{scope}[rotate=45, thick, red]
    \draw (0,0) rectangle (1,1);
  \end{scope}
  \foreach \x in {0,1,2} \draw (\x,0) -- (\x,1);
  \foreach \x in {0,...,3} { \draw (\x,0) circle (0.1); }
  \foreach \x/\y in {0/a, 1/b} \node at (\x,0) {\y};
  \foreach \i [count=\n] in {0.5,1,...,2} \node at (\i,0) {\n};
  \node (n) at (0,5) {n};
  \node[right=of n] (m) {m};
  \draw[->] (n) -- (m);
  \coordinate (A) at (0,0);
  \coordinate (B) at (4,2);
  \draw ($(A)!0.5!(B)$) circle (0.1);
  \draw ($(A)!1cm!(B)$) circle (0.1);
  \draw ($(A)+(1,0)$) -- ($2*(B)-(A)$);
  \draw ($(A)!0.5!90:(B)$) -- (B);
  \draw ($(A)!(2,3)!(B)$) circle (0.1);
\end{tikzpicture}
